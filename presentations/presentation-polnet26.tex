% ------------------------------------------------------------------ 2 SETUP
\begin{frame}{A natural quasi-experiment}
\begin{itemize}
  \item In an ordinary legislature, today's collaboration network is the sediment of decades: careers, committees, party discipline.
  \vspace{0.25cm}
  \item The Chilean Constitutional Convention (Jul 2021 -- Jul 2022) started from \emphr{zero relational stock}: a majority of newcomers and independents, 17 reserved indigenous seats, a body that dissolved on delivery.
  \vspace{0.25cm}
  \item Each constitutional initiative required \emphr{8--16 sponsors}: signing = forming a visible coalition. 
  \vspace{0.25cm}
  \item Each norm needed \emphr{2/3 of the floor (103 of 154)}, so being born was cheap, but surviving was expensive.
\end{itemize}
\end{frame}

% ------------------------------------------------------------------ 7 ERGM TOOL
\begin{frame}{Bipartite ERGM}
\begin{itemize}
  \item Unit of analysis: the \emphr{person $\times$ initiative} signature ($y_{ia} = 1$ if member $i$ signed initiative $a$).
\end{itemize}
{\small $$\Pr(Y = y) = \frac{\exp\big(\theta^\top s(y)\big)}{\kappa(\theta)}$$}
{\small $$
s(y) = \Big(\underbrace{\sum_{i,a} y_{ia}}_{\text{edges}},\;
\underbrace{\sum_{a}\sum_{i<j} y_{ia} y_{ja}\,\mathbf 1\{b_i,b_j\}{=}\{g,h\}}_{\text{bloc mixing: 15 cells}},\;
\underbrace{\sum_{a}\sum_{i<j} y_{ia} y_{ja}\,\mathbf 1[x_i = x_j]\,\mathbf 1[b_i = b_j = g]}_{\text{profile homophily within bloc } g},\;\dots\Big)$$}
\begin{itemize}
  \item Variables: per-document ranges of $\theta_1, \theta_2$, age, education; the bloc contingent's own $\theta_1$ range; \texttt{gwb1degree}; \texttt{gwdsp}. 
  \item Seven models, one per commission.
  \item Estimation: MPLE as a logistic on change statistics; SEs by \emphr{initiative bootstrap} (B = 500).
\end{itemize}
\end{frame}

% ------------------------------------------------------------------ 10 RQ2a DESIGN
\begin{frame}{RQ2a design: the norms era, and a careful clock}
\begin{itemize}
  \item Everything here uses \emphr{only the norms era} (Feb 15 -- May 14: content voted under the 2/3 rule) and $\theta$ re-estimated with era votes only — a regime-homogeneous thermometer.
\end{itemize}
$$E_{i,t} = \frac{\sum_{j \neq i} w_{ij,t}\,\theta_{j,t}}{\sum_{j \neq i} w_{ij,t}}
\qquad\qquad
\Delta\theta_{i,t} = \alpha_i + \mu_t + \beta\,\theta_{i,t-1} + \lambda\,E_{i,t-1} + \lambda_{innov}\,\tilde E_{i,t+2} + \varepsilon_{it}$$
\begin{itemize}
  \item $\alpha_i$: person fixed effects (only within-person variation); $\mu_t$: wave-date fixed effects (common shocks); $\lambda > 0$ = I move toward where my neighborhood \emph{was}.
  \item \emphr{The clock}: positions at $t{+}1$ are built from votes cast at $t$. So the lag uses $E_{i,t-1}$ (votes through $t{-}2$), and the ``strictly future'' innovation $\tilde E_{i,t+2}$ uses $t{+}2$ — never $t{\pm}1$.
  \item $\tilde E_{i,t+2}$: the part of future exposure the past did not contain. If ``influence'' were really selection — signing with people I'm already moving toward — the future should predict and the lag should die.
\end{itemize}
\end{frame}

\begin{frame}{RQ2b: defection travels along co-sponsorship ties}
\begin{itemize}
  \item Defection: in roll call $v$, voting against the modal vote of one's own bloc. Exposure: the $w$-weighted share of $i$'s co-signers defecting \emph{in the same roll call}:
\end{itemize}
$$X_{iv} = \frac{\sum_{j \neq i} w_{ij}\, D_{jv}}{\sum_{j \neq i} w_{ij}},
\qquad
\Pr(D_{iv} = 1) = \Lambda\big(\eta_i + \mu_v + \phi\, X_{iv}\big)$$
\begin{center}\small
\begin{tabular}{lcc}
\toprule
 & $\phi$ & $p$ \\
\midrule
Exposure to defectors (observed) & $+11.2$ & $<10^{-100}$ \\

\end{itemize}

\sectionslide{RQ3b — Members}{Whose success is it?}

\begin{frame}{Does the context an initiative is born into matter?}
\begin{itemize}
  \item Each member's \emphr{success}: of all the article text you co-signed, how much survived into the draft? ($y_i$ = mean textual retention of $i$'s articles, 0 to 1.)
  \vspace{0.3cm}
  \item The raw fact: success is \emphr{not individual} — it clusters on the co-sponsorship network (Moran's $I = 0.44$, $p \approx 10^{-163}$). Your success looks like your co-signers' success.
  \vspace{0.3cm}
  \item The model that takes that seriously — each member's success related to the \emph{average success of their co-signers}, their own attributes, and their co-signers' attributes:
\end{itemize}
$$y = \rho\, W y + X\beta + W X \gamma + \varepsilon, \qquad W = \text{row-normalized co-sponsorship network}$$
\begin{itemize}
  \item One number, $\rho$, answers the title: does the company an initiative is born into matter?
\end{itemize}
\end{frame}

\begin{frame}{Reading $\rho$: a spillover, up to identification}
\begin{itemize}
  \item This equation is not ad hoc: it is the \emphr{equilibrium of an effort game} (Battaglini et al., AJPS 2020). If my effectiveness rises with my allies' effectiveness, the unique equilibrium makes everyone's success equal their \emphr{Katz--Bonacich centrality} — and the reduced form is exactly this model.
  \vspace{0.3cm}
  \item Under that reading — \emphr{up to an identification test currently in progress} (instrumenting the network with pre-Convention ties; the design is ready) — $\rho$ is the \emphr{coalition-effectiveness spillover}: connected members' effectiveness spills over into yours.
  \vspace{0.3cm}
  \item And the covariates tell you what the ``context'' is made of: your own distance to the pivot doesn't matter ($p=.95$) — your \emph{co-signers'} distance does ($-0.27$). \emphr{The context is the coalition you stand in.}
  \vspace{0.3cm}
  \item Taken together: the evidence supports the network view of success — where you stand shapes how well you do, beyond who you are.
\end{itemize}
\end{frame}

% ------------------------------------------------------------------ 18 TAKEAWAYS
\begin{frame}{Takeaways}
\begin{itemize}
  \item The network \emphr{was predictable} from pre-existing traits: bloc, 2-D ideology, and — conditionally — territory. Credentials organized nothing, in any bloc.
  \item The bloc partition pays: territory is the right's internal glue and the system's bridge; gender crosses blocs; experience lives in the left; \texttt{gwdsp} $< 0$ everywhere — the structural signature of a tabula rasa.
  \item Positions barely move: \emphr{selection}, with one dated exception at the onset of the two-thirds regime.
  \item Behavior does travel: defection clusters along co-sponsorship ties at twice the mechanical rate, carried by newcomers.
  \item Texts win by \emphr{geometry and team history}, not credentials; and the ``context'' of success is the coalition itself.
\end{itemize}
\vspace{0.2cm}

\end{frame}
