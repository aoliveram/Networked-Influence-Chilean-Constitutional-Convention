% ------------------------------------------------------------------ 2 SETUP
\begin{frame}{A natural quasi-experiment}
\begin{itemize}
  \item In an ordinary legislature, today's collaboration network is the sediment of decades: careers, committees, party discipline. Any estimate of ``what organizes collaboration'' is contaminated by that history.
  \vspace{0.25cm}
  \item The Chilean Constitutional Convention (Jul 2021 -- Jul 2022) started from \emphr{zero relational stock}: a majority of newcomers and independents, 17 reserved indigenous seats, a body that dissolved on delivery — no electoral shadow of the future.
  \vspace{0.25cm}
  \item Two rules structure everything. Each constitutional initiative required \emphr{8--16 sponsors}: signing = forming a visible, dated coalition. Each norm needed \emphr{two thirds of the floor (103 of 154)}: being born was cheap, surviving was expensive.
\end{itemize}
\vspace{0.3cm}
\nota{A note on standpoint: I am a methodologist coming from the natural sciences. I report what the models find and comment on some of it; the political reading is ongoing joint work with Jorge F\'abrega (sociologist, UDD).}
\end{frame}

% ------------------------------------------------------------------ 5 IDEOLOGY
\begin{frame}{Pre-network ideology: measured before the network existed}
\begin{itemize}
  \item \emphr{Pre-network positions}: 2-D W-NOMINATE on the \emph{first month} of floor votes only (147 roll calls, Jul--Aug 2021), replicating F\'abrega (2022).
  \begin{itemize}
    \item That window predates the commissions (Oct), the initiatives (Nov) and the operative two-thirds rule (Feb) — so $\theta$ is exogenous to the network we study.
    \item Why 2-D: the second dimension separates the reserved indigenous seats from the classic left--right cleavage; correct classification rises from 89.4\% to 91.6\%.
  \end{itemize}
  \vspace{0.3cm}
  \item \emphr{Dynamic revealed positions}: dynamic IRT over all 4,707 roll calls $\to$ trajectories $\theta_{i,t}$ over 91 periods,
  $$\Pr(\text{yes}_{iv}) = \Phi\big(\beta_v\,\theta_{i,t(v)} - \alpha_v\big), \qquad \theta_{i,t} \sim \mathcal N(\theta_{i,t-1}, \omega^2).$$
\end{itemize}
\end{frame}

% ------------------------------------------------------------------ 6 RQs
\begin{frame}{Research questions}
\begin{itemize}
  \item \emphr{RQ1 — Formation}: with zero relational stock, could the co-sponsorship network be predicted from what people \emph{brought with them} (ideology, bloc, district, profile)?
  \vspace{0.3cm}
  \item \emphr{RQ2a — Positions}: does network exposure \emph{move} members' ideological positions, or do members select into like-minded coalitions?
  \vspace{0.3cm}
  \item \emphr{RQ2b — Behavior}: does voting \emph{defection} travel along co-sponsorship ties?
  \vspace{0.3cm}
  \item \emphr{RQ3 — Success}: what makes an article survive into the draft — and does the \emph{context} an initiative is born into matter?
\end{itemize}
\end{frame}

% ------------------------------------------------------------------ 7 ERGM TOOL
\begin{frame}{The tool: a hybrid bipartite ERGM, by commission and by bloc}
\begin{itemize}
  \item Unit of analysis: the \emphr{person $\times$ initiative} signature ($y_{ia} = 1$ if member $i$ signed initiative $a$). No projection — one signature counts once.
\end{itemize}
{\small $$\Pr(Y = y) = \frac{\exp\big(\theta^\top s(y)\big)}{\kappa(\theta)},\qquad
s(y) = \Big(\underbrace{\sum_{i,a} y_{ia}}_{\text{edges}},\;
\underbrace{\sum_{a}\sum_{i<j} y_{ia} y_{ja}\,\mathbf 1\{b_i,b_j\}{=}\{g,h\}}_{\text{bloc mixing: 15 cells}},\;
\underbrace{\sum_{a}\sum_{i<j} y_{ia} y_{ja}\,\mathbf 1[x_i = x_j]\,\mathbf 1[b_i = b_j = g]}_{\text{profile homophily within bloc } g},\;\dots\Big)$$}
\begin{itemize}
  \item Separating by bloc \emphr{never cuts the network} (that would mutilate coalitions): the network stays whole and the \emph{counters} specialize by bloc. Each homophily counter partitions exactly into 5 within-bloc + 1 across-bloc terms.
  \item Plus: per-document ranges of $\theta_1, \theta_2$, age, education; the bloc contingent's own $\theta_1$ range; \texttt{gwb1degree}; \texttt{gwdsp}. Seven models, one per commission (pooling inverts homophily signs — Simpson's paradox, verified).
  \item Estimation: MPLE as a logistic on change statistics (certified against \texttt{ergm} to $10^{-11}$); SEs by \emphr{initiative bootstrap} (B = 500: resample initiatives, rebuild network, re-estimate).
\end{itemize}
\end{frame}

% ------------------------------------------------------------------ 10 RQ2a DESIGN
\begin{frame}{RQ2a design: exposure and within-person change}
\begin{itemize}
  \item Exposure of $i$ at wave $t$: mean position of co-signers, weighted by accumulated collaboration $w_{ij,t}$; the model uses only \emph{within-person} variation:
\end{itemize}
$$E_{i,t} = \frac{\sum_{j \neq i} w_{ij,t}\,\theta_{j,t}}{\sum_{j \neq i} w_{ij,t}}
\qquad\qquad
\Delta\theta_{i,t} = \alpha_i + \beta\,\theta_{i,t-1} + \lambda\,E_{i,t-1} + \varepsilon_{it}$$
\begin{itemize}
  \item Individual fixed effects $\alpha_i$; SEs clustered by member; 4,355 person-wave observations.
  \item If the network drags positions, $\lambda > 0$. Result: $\hat\lambda = +0.007$ (SE $0.004$) — \emphr{null}.
\end{itemize}
\end{frame}

% ------------------------------------------------------------------ 13 ERA 2/3
\begin{frame}{A dated exception: the norms era, measured with its own thermometer}
\begin{itemize}
  \item Clean test: decompose future exposure into its predictable part and its \emph{innovation} $\tilde E_{i,t+1}$ (the within-residual of lead on lag), and race them:
\end{itemize}
$$\Delta\theta_{i,t} = \alpha_i + \beta\,\theta_{i,t-1} + \lambda_{\text{lag}}\,E_{i,t-1} + \lambda_{\text{innov}}\,\tilde E_{i,t+1} + \varepsilon_{it}$$
\begin{center}\small
\begin{tabular}{lccc}
\toprule
Thermometer & Lag $\lambda_{\text{lag}}$ & Innovation $\lambda_{\text{innov}}$ & N \\
\midrule

  \item Reading: selection is the rule; a bounded, dated accommodation appears when the 103-vote arithmetic was new. Lag and innovation \emph{coexist} — not clean causality.
\end{itemize}
\end{frame}

% ------------------------------------------------------------------ 14 RQ2b
\begin{frame}{RQ2b: defection travels along co-sponsorship ties}
\begin{itemize}
  \item Defection: in roll call $v$, voting against the modal vote of one's own bloc. Exposure: the $w$-weighted share of $i$'s co-signers defecting \emph{in the same roll call}:
\end{itemize}
$$X_{iv} = \frac{\sum_{j \neq i} w_{ij}\, D_{jv}}{\sum_{j \neq i} w_{ij}},
\qquad
\Pr(D_{iv} = 1) = \Lambda\big(\eta_i + \mu_v + \phi\, X_{iv}\big)$$
\begin{center}\small
\begin{tabular}{lcc}
\toprule
 & $\phi$ & $p$ \\
\midrule
Exposure to defectors (observed) & $+11.2$ & $<10^{-100}$ \\
Expected mechanically (permutation within bloc $\times$ vote, 200 draws) & $+6.0$ [p95: $6.1$] & $<0.005$ \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.2cm}
\begin{itemize}
  \item Half the raw effect is mechanical; \emphr{what remains is large}, travels more strongly \emph{across} commissions, and leaves a short echo that dies within days. Carried by the \emph{newcomer} pairs, not the old guard.
\end{itemize}
\end{frame}

\end{frame}

% ------------------------------------------------------------------ 17 SDM
\begin{frame}{Does the context an initiative is born into matter?}
\begin{itemize}
  \item Member-level success: $y_i$ = mean textual retention of the articles $i$ co-signed. It clusters on the co-sponsorship network (Moran's $I = 0.44$). The spatial Durbin model asks whether that context matters:
\end{itemize}
$$y = \rho\, W y + X\beta + W X \gamma + \varepsilon, \qquad W = \text{row-normalized co-sponsorship network}$$
\begin{itemize}
  \item $\rho = 0.891$ ($p < 10^{-15}$; robust $0.63$--$0.95$ across $W$ variants). So ``context matters'' — \emphr{which by itself explains nothing}. The question is \emphr{what the context is}.
  \vspace{0.2cm}
  \item Decomposition: the neighborhood's success $Wy$ is predictable from one's \emph{own} covariates ($R^2 = 0.81$); and the only covariate that matters — distance to the $\tfrac{2}{3}$ pivot — \emphr{migrates entirely from the person to the neighborhood} when moving from OLS to the Durbin model (own: $p = .95$; neighborhood's: $-0.27$, $p < .05$).
  \vspace{0.2cm}
  \item Answer: \emphr{the context is the coalition you stand in} — its composition and its position in the coalition landscape. Success is a coalition good, not diffusion; $\rho$ is read as clustering, not influence.
\end{itemize}
\end{frame}

% ------------------------------------------------------------------ 18 TAKEAWAYS
\begin{frame}{Takeaways}
\begin{itemize}
  \item The network \emphr{was predictable} from pre-existing traits: bloc, 2-D ideology, and — conditionally — territory. Credentials organized nothing, in any bloc.
  \item The bloc partition pays: territory is the right's internal glue and the system's bridge; gender crosses blocs; experience lives in the left; \texttt{gwdsp} $< 0$ everywhere — the structural signature of a tabula rasa.
  \item Positions barely move: \emphr{selection}, with one dated exception at the onset of the two-thirds regime.
  \item Behavior does travel: defection clusters along co-sponsorship ties at twice the mechanical rate, carried by newcomers.
  \item Texts win by \emphr{geometry and team history}, not credentials; and the ``context'' of success is the coalition itself.
\end{itemize}
\vspace{0.2cm}
\nota{Political interpretation: ongoing joint work with Jorge F\'abrega. Preliminary conclusions — the analysis continues.}
\end{frame}
